set(_m3 "${CMAKE_SOURCE_DIR}/dep/wasm3/source")
set(_sources bind code compile core emit env exec function info module parse)
set(_files)
foreach(_s IN LISTS _sources)
  list(APPEND _files "${_m3}/m3_${_s}.c")
endforeach()
add_library(aeternum-wasm3 STATIC ${_files})
target_include_directories(aeternum-wasm3 PUBLIC "${_m3}")
target_compile_definitions(aeternum-wasm3 PRIVATE d_m3MaxLinearMemoryPages=16)
set_target_properties(aeternum-wasm3 PROPERTIES C_STANDARD 99 FOLDER "dep")
# wasm3 uses standard variadic macro expansion; legacy MSVC preprocessor is invalid.
if(MSVC)
  target_compile_options(aeternum-wasm3 PRIVATE /Zc:preprocessor)
endif()
add_library(aeternum-wasm STATIC Sandbox.cpp Allocator.cpp Pipeline.cpp)
target_include_directories(aeternum-wasm PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
target_link_libraries(aeternum-wasm PRIVATE aeternum-wasm3 PUBLIC aeternum-profiler)
set_target_properties(aeternum-wasm PROPERTIES FOLDER "server")

if(MSVC)
  target_compile_options(aeternum-wasm PRIVATE /Zc:preprocessor)
endif()
